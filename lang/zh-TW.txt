#  WARDEN 语言包 —— 简体中文

#  【怎么用】
#    这个文件是纯文本，用记事本就能改。
#    想加一门语言：复制一份改成 en.txt 放进
#      %LOCALAPPDATA%\WARDEN\lang\
#    程序会自动认出来，不用重新编译。
#
#  【格式】
#    [meta]      语言信息       key=value
#    [ui]        界面文字       key=value
#    [log.tag]   日志前缀名     key=value
#    [log.rule]  日志整行翻译   正则<TAB>译文
#               译文里 {1} {2} 引用正则的捕获组
#               规则按顺序匹配，第一条命中的生效
#    以 # 开头是注释
#
#  【★ 请勿改 key，只改等号/制表符右边的内容】
# ============================================================================
